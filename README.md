```python
import packagexyz

class Circle{  
    const PI_6DEC = 3.141592;
    public function circumference($radius){ 
        return 2 * self::PI_6DEC * $radius;
    }
    public function getArea($radius){ 
        return self::PI_6DEC * $radius * $radius;
    }
}

$circle = new Circle();
$given_radius = 10;
$circle_area = $circle->getArea($given_radius);
```
